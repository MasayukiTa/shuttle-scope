# SECURITY SCOPE
# This manifest is CI-only compatibility tooling for exporting the immutable,
# SHA-256-pinned upstream TrackNet TensorFlow checkpoint. It is not installed by
# backend/requirements.txt and is not part of the production runtime.
# The workflow has contents:read only, persists no checkout credential, and
# processes only allowlisted/hash-verified checkpoint and test-vector bytes.
#
# TrackNet TensorFlow-checkpoint -> ONNX export smoke only.
# TensorFlow 2.15.1 is the final Keras 2 line used by the upstream TrackNet
# checkpoint and therefore requires Python <=3.11 and NumPy <2.
numpy==1.26.4
# TensorFlow 2.15 accepts protobuf >=3.20.3,<5. tf2onnx 1.17 removed the
# old ~=3.20 cap, so 4.25.8 closes the protobuf DoS advisory that is
# patchable within the supported Keras 2 exporter stack.
protobuf==4.25.8
tensorflow==2.15.1
tf2onnx==1.17.0
# ONNX 1.18.0 is the newest converter-compatible release that still works
# with TensorFlow 2.15 / ml-dtypes~=0.3.1. Keep it explicit here so the
# workflow installs from one pinned manifest rather than an inline pip command.
onnx==1.18.0
onnxruntime==1.17.3
opencv-python-headless==4.10.0.84
